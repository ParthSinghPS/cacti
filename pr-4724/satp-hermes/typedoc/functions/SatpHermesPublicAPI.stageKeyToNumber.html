    • stageKeyToNumber(stageKey: SatpStageKey): undefined | number

      • stageKey: SatpStageKey

        The string stage key (e.g., "stage0", "stage1", "crash")

      Returns undefined | number

      The numeric stage value (0-3) or undefined for crash
